CREATE TABLE SALES_REP
(
	REP_ID CHAR (2) PRIMARY KEY,
    FIRST_NAME CHAR(20) NOT NULL,
    LAST_NAME CHAR (20) NOT NULL,
    ADDRESS CHAR (20),
    CITY CHAR (15),
    STATE CHAR(2),
    POSTAL CHAR (5),
    CELL_PHONE CHAR(12),
    COMMISSION DECIMAL (7, 2),

CREATE TABLE `CUSTOMER` (
  `CUST_ID` char(3) NOT NULL,
  `FIRST_NAME` varchar(20) DEFAULT NULL,
  `LAST_NAME` varchar(20) DEFAULT NULL,
  `ADDRESS` varchar(20) DEFAULT NULL,
  `CITY` varchar(15) DEFAULT NULL,
  `STATE` char(2) DEFAULT NULL,
  `POSTAL` char(5) DEFAULT NULL,
  `EMAIL` varchar(30) DEFAULT NULL,
  `BALANCE` decimal(7,2) DEFAULT NULL,
  `CREDIT_LIMIT` decimal(7,2) DEFAULT NULL,
  `REP_ID` char(2) DEFAULT NULL,
  PRIMARY KEY (`CUST_ID`)

CREATE TABLE `INVOICE_LINE` (
  `INVOICE_NUM` char(5) NOT NULL,
  `ITEM_ID` char(4) NOT NULL,
  `QUANTITY` decimal(10,0) DEFAULT NULL,
  `QUOTED_PRICE` decimal(7,2) DEFAULT NULL,
  PRIMARY KEY (`INVOICE_NUM`,`ITEM_ID`)

CREATE TABLE `ITEM` (
  `ITEM_ID` char(4) NOT NULL,
  `DESCRIPTION` varchar(30) DEFAULT NULL,
  `ON_HAND` decimal(10,0) DEFAULT NULL,
  `CATEGORY` char(3) DEFAULT NULL,
  `LOCATION` char(1) DEFAULT NULL,
  `PRICE` decimal(7,2) DEFAULT NULL,
  PRIMARY KEY (`ITEM_ID`)

CREATE TABLE `SALES_REP` (

  `CELL_PHONE` char(12) DEFAULT NULL,
  `COMMISSION` decimal(7,2) DEFAULT NULL,
  `RATE` decimal(3,2) DEFAULT NULL,
  PRIMARY KEY (`REP_ID`)

CREATE PROCEDURE GET_CREDIT_LIMIT(IN I_CUST_ID CHAR(3))
BEGIN 

    DECLARE I_CUSTOMER_NAME VARCHAR(41);
    DECLARE I_CREDIT_LIMIT DECIMAL(10,2);

    SELECT CONCAT(FIRST_NAME, ' ', LAST_NAME), CREDIT_LIMIT
        INTO I_CUSTOMER_NAME, I_CREDIT_LIMIT
        FROM CUSTOMER
        WHERE CUST_ID = I_CUST_ID;
    
    SELECT I_CUSTOMER_NAME, I_CREDIT_LIMIT;

END //

CREATE PROCEDURE GET_INVOICE_DATE(IN I_INVOICE_NUM CHAR(5))
BEGIN 

    DECLARE I_CUST_ID CHAR(3);
    DECLARE I_CUST_NAME VARCHAR(41);
    DECLARE I_INVOICE_DATE DATE;

    SELECT INVOICES.CUST_ID, CONCAT(FIRST_NAME,' ',LAST_NAME), INVOICE_DATE
        INTO I_CUST_ID, I_CUST_NAME, I_INVOICE_DATE
        FROM CUSTOMER, INVOICES
        WHERE INVOICES.CUST_ID = CUSTOMER.CUST_ID AND INVOICE_NUM = I_INVOICE_NUM;
    
    SELECT I_CUST_ID, 
           I_CUST_NAME,
           I_INVOICE_DATE;

END //

        SELECT CONCAT("No date with this INVOICE found:", I_INVOICE_NUM) AS MESSAGE;

        SET INVOICE_DATE = I_INVOICE_DATE
        WHERE INVOICE_NUM = I_INVOICE_NUM;

CREATE TRIGGER ADD_COMMISSION_TRIG
    AFTER INSERT ON SALES_REP FOR EACH ROW 
BEGIN 
    UPDATE SALES_REP
        SET COMMISSION = COMMISSION * CUSTOMER.BALANCE 
    WHERE CUSTOMER.REP_ID = SALES_REP.REP_ID;

END //

DELIMITER ;


DELIMITER //
CREATE TRIGGER UPD_COMMISSION_TRIG
    AFTER UPDATE ON CUSTOMER FOR EACH ROW 
BEGIN 
    UPDATE CUSTOMER
        SET BALANCE = (NEW.BALANCE - OLD.BALANCE) * SALES_REP.RATE
    WHERE CUSTOMER.REP_ID = SALES_REP.REP_ID;

END //

DELIMITER ;

DELIMITER //
CREATE TRIGGER DEL_COMMISSION_TRIG
    AFTER UPDATE ON CUSTOMER FOR EACH ROW 
BEGIN 
    UPDATE CUSTOMER
        SET BALANCE = (BALANCE * SALES_REP.RATE) - SALES_REP.COMMISSION
    WHERE CUSTOMER.REP_ID = SALES_REP.REP_ID;

END //

DELIMITER ;
